1. Activity Model

The Activity entity represents an employee's daily activity. It stores the employee, date, description, start time, expected end time and current status. 


package com.hdfcbank.tracker.model;

import jakarta.persistence.*;
import lombok.*;
import java.time.LocalDate;
import java.time.LocalTime;

@Entity
@Table(name = "activities")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class Activity {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.EAGER)
    @JoinColumn(name = "user_id", nullable = false)
    private User user;

    @Column(nullable = false)
    private LocalDate date;

    @Column(nullable = false, length = 500)
    private String description;

    @Column(name = "start_time", nullable = false)
    private LocalTime startTime;

    @Column(name = "expected_end_time", nullable = false)
    private LocalTime expectedEndTime;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false)
    private ActivityStatus status;
}

3. Activity Repository

The repository provides database operations for Activity objects. 


package com.hdfcbank.tracker.repository;

import com.hdfcbank.tracker.model.Activity;
import com.hdfcbank.tracker.model.User;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.stereotype.Repository;

import java.time.LocalDate;
import java.util.List;

@Repository
public interface ActivityRepository
        extends JpaRepository<Activity, Long> {

    List<Activity> findByUserAndDateBetween(
            User user,
            LocalDate start,
            LocalDate end);

    List<Activity> findByDateBetween(
            LocalDate start,
            LocalDate end);

    List<Activity> findByUser(User user);
}

4. Activity Service

The service layer contains the business logic for activities.

Main responsibilities: calculating weeks, retrieving activities, adding activities, updating activities and deleting activities. 

@Service
@RequiredArgsConstructor
public class ActivityService {

    private final ActivityRepository activityRepository;
    private final UserRepository userRepository;

    public LocalDate getWeekStart(LocalDate date) {

        return date.with(
            TemporalAdjusters.previousOrSame(
                DayOfWeek.MONDAY
            )
        );
    }

    public LocalDate getWeekEnd(LocalDate date) {

        return date.with(
            TemporalAdjusters.nextOrSame(
                DayOfWeek.SUNDAY
            )
        );
    }

    @Transactional(readOnly = true)
    public List<User> getAllEmployees() {

        return userRepository.findByRole(
            Role.EMPLOYEE
        );
    }

    @Transactional(readOnly = true)
    public List<Activity> getUserWeeklyActivities(
            Long userId,
            LocalDate weekDate) {

        User user = userRepository.findById(userId)
            .orElseThrow(() ->
                new RuntimeException("User not found"));

        return activityRepository
            .findByUserAndDateBetween(
                user,
                getWeekStart(weekDate),
                getWeekEnd(weekDate)
            );
    }

    @Transactional
    public String addActivity(
            String adid,
            LocalDate date,
            String description,
            LocalTime startTime,
            LocalTime expectedEndTime,
            ActivityStatus status) {

        User user = userRepository.findByAdid(adid)
            .orElseThrow(() ->
                new RuntimeException("User not found"));

        if (expectedEndTime.isBefore(startTime)) {

            return "error:Expected end time must be after start time";
        }

        Activity activity = Activity.builder()
            .user(user)
            .date(date)
            .description(description)
            .startTime(startTime)
            .expectedEndTime(expectedEndTime)
            .status(status)
            .build();

        activityRepository.save(activity);

        return "success:Activity added successfully";
    }
}

5. Activity Controller

The controller handles HTTP requests and sends the required data to the Thymeleaf pages. 


@Controller
@RequestMapping("/activity")
public class ActivityController
        extends BaseController {

    private final ActivityService activityService;

    @GetMapping
    public String activityList(
            @RequestParam(required = false)
            String week,
            Authentication authentication,
            Model model) {

        loadUserToModel(authentication, model);

        model.addAttribute(
            "activePage",
            "activity"
        );

        LocalDate weekDate =
            (week != null && !week.isBlank())
                ? LocalDate.parse(week)
                : LocalDate.now();

        model.addAttribute(
            "employees",
            activityService.getAllEmployees()
        );

        return "activity/index";
    }

    @GetMapping("/my")
    public String myActivities(
            @RequestParam(required = false)
            String week,
            Authentication authentication,
            Model model) {

        loadUserToModel(authentication, model);

        model.addAttribute(
            "activePage",
            "activity"
        );

        return "activity/my-activities";
    }
}
