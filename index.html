// Simpler — just clone first row
function addMemberRow() {
    const container =
            document.getElementById('memberRows');
    const rows = container.querySelectorAll(
            '.member-assign-row');
    const firstRow = rows[0];
    const newRow = firstRow\.cloneNode(true);

    // Reset values
    newRow\.querySelector(
            'select[name="memberUserIds"]').value = '';
    newRow\.querySelector(
            'input[name="stageNames"]').value =
            defaultStages[stageIndex]
            || 'Stage ' + (stageIndex + 1);
    newRow\.querySelector(
            'input[name="stageDescriptions"]').value = '';

    stageIndex++;
    container.appendChild(newRow);
}

function removeRow(btn) {
    const rows = document.querySelectorAll(
            '.member-assign-row');
    if (rows.length <= 1) {
        alert('At least one member is required');
        return;
    }
    btn.closest('.member-assign-row').remove();
}

function toggleCustomCategory(select) {
    const customInput =
            document.getElementById(
                    'customCategoryInput');
    if (select.value === 'OTHER') {
        customInput.style.display = 'block';
        customInput.required = true;
    } else {
        customInput.style.display = 'none';
        customInput.required = false;
        customInput.value = '';
    }
}
