    @GetMapping("/reports/export")
    public void exportCsv(
    @RequestParam(required = false) String month,
    @RequestParam(required = false) Long vendorId,
    HttpServletResponse response) throws IOException {

    YearMonth selectedMonth = (month != null && !month.isBlank())
    ? YearMonth.parse(month)
    : YearMonth.now();

    List<Attendance> records = (vendorId != null)
        ? adminReportService.getMonthlyAttendanceByVendor(
        selectedMonth, vendorId)
        : adminReportService.getMonthlyAttendance(
        selectedMonth);

        response.setContentType("text/csv");
        response.setHeader("Content-Disposition",
        "attachment; filename=attendance-"
        + selectedMonth + ".csv");

        PrintWriter writer = response.getWriter();
        writer.println(
        "Employee Name,Emp Code,Department,"
        + "Vendor,Date,Check In,Check Out,"
        + "Total Hours,Status");

        for (Attendance att : records) {
        writer.println(
        escape(att.getUser().getFullName()) + ","
        + escape(att.getUser().getEmpCode()) + ","
        + escape(att.getUser().getDepartment() != null
        ? att.getUser().getDepartment() : "—") + ","
        + escape(att.getUser().getVendor() != null
        ? att.getUser().getVendor().getName() : "—") + ","
        + att.getDate() + ","
        + (att.getCheckIn() != null
        ? String.format("%02d:%02d",
        att.getCheckIn().getHour(),
        att.getCheckIn().getMinute())
        : "—") + ","
        + (att.getCheckOut() != null
        ? String.format("%02d:%02d",
        att.getCheckOut().getHour(),
        att.getCheckOut().getMinute())
        : "—") + ","
        + (att.getTotalHours() != null
        ? att.getTotalHours() : "0") + ","
        + att.getStatus().name()
        );
        }
        writer.flush();
        }

        @GetMapping("/reports/export-monthly")

        @RequestParam(required = false) String month,
        HttpServletResponse response) throws IOException {

        : YearMonth.now();

                "OpenSource", "OpenShift");

                    adminReportService
                    .getEmployeesByDepartment(dept);

                    // ── Header style ────────────────────────────

                    workbook.createCellStyle();
                    headerStyle.setFillForegroundColor(
                    new XSSFColor(new byte[]{
                    (byte)14,(byte)27,(byte)46},
                    null));
                    headerStyle.setFillPattern(
                    FillPatternType.SOLID_FOREGROUND);
                    XSSFFont headerFont = workbook.createFont();
                    headerFont.setColor(new XSSFColor(new byte[]{
                    (byte)228,(byte)233,(byte)240}, null));
                    headerFont.setBold(true);
                    headerStyle.setFont(headerFont);

                    // ── Header row ──────────────────────────────
                    Row header = sheet.createRow(0);
                    Cell dateCell = header.createCell(0);
                    dateCell.setCellValue("DATE");
                    dateCell.setCellStyle(headerStyle);
                    sheet.setColumnWidth(0, 5000);

                    for (int i = 0; i < deptTeam.size(); i++) {
                    Cell empCell = header.createCell(i + 1);
                    empCell.setCellValue(
                    deptTeam.get(i).getFullName()
                    + " (" + deptTeam.get(i).getEmpCode()
                    + ")");
                    empCell.setCellStyle(headerStyle);
                    sheet.setColumnWidth(i + 1, 4500);
                    }

                    // ── Cell styles ─────────────────────────────
                    XSSFCellStyle presentStyle =
                    workbook.createCellStyle();
                    presentStyle.setFillForegroundColor(
                    new XSSFColor(new byte[]{
                    (byte)225,(byte)236,(byte)228},
                    null));
                    presentStyle.setFillPattern(
                    FillPatternType.SOLID_FOREGROUND);

                    XSSFCellStyle lateStyle =
                    workbook.createCellStyle();
                    lateStyle.setFillForegroundColor(
                    new XSSFColor(new byte[]{
                    (byte)234,(byte)226,(byte)208},
                    null));
                    lateStyle.setFillPattern(
                    FillPatternType.SOLID_FOREGROUND);

                    XSSFCellStyle absentStyle =
                    workbook.createCellStyle();
                    absentStyle.setFillForegroundColor(
                    new XSSFColor(new byte[]{
                    (byte)240,(byte)220,(byte)222},
                    null));
                    absentStyle.setFillPattern(
                    FillPatternType.SOLID_FOREGROUND);

                    XSSFCellStyle weekendStyle =
                    workbook.createCellStyle();
                    weekendStyle.setFillForegroundColor(
                    new XSSFColor(new byte[]{
                    (byte)220,(byte)220,(byte)220},
                    null));
                    weekendStyle.setFillPattern(
                    FillPatternType.SOLID_FOREGROUND);

                    XSSFCellStyle leaveStyle =
                    workbook.createCellStyle();
                    leaveStyle.setFillForegroundColor(
                    new XSSFColor(new byte[]{
                    (byte)222,(byte)229,(byte)239},
                    null));
                    leaveStyle.setFillPattern(
                    FillPatternType.SOLID_FOREGROUND);

                    XSSFCellStyle wfhStyle =
                    workbook.createCellStyle();
                    wfhStyle.setFillForegroundColor(
                    new XSSFColor(new byte[]{
                    (byte)220,(byte)235,(byte)220},
                    null));
                    wfhStyle.setFillPattern(
                    FillPatternType.SOLID_FOREGROUND);

                    // ── Data rows ───────────────────────────────
                    int rowNum = 1;
                    for (int day = 1;
                    day <= yearMonth.lengthOfMonth(); day++) {

                    LocalDate date = yearMonth.atDay(day);
                    boolean isWeekend =
                    date.getDayOfWeek() == DayOfWeek.SATURDAY
                    || date.getDayOfWeek() == DayOfWeek.SUNDAY;

                    Row row = sheet.createRow(rowNum++);

                    Cell dc = row.createCell(0);
                    dc.setCellValue(
                    date.getDayOfMonth() + " "
                    + date.getMonth().name().substring(0, 3)
                    + " ("
                    + date.getDayOfWeek().name()
                    .substring(0, 3) + ")");
                    if (isWeekend)
                    dc.setCellStyle(weekendStyle);

                    for (int i = 0; i < deptTeam.size(); i++) {
                    com.hdfcbank.tracker.model.User emp =
                    deptTeam.get(i);
                    LocalDate finalDate = date;
                    Cell cell = row.createCell(i + 1);

                    if (isWeekend) {
                    cell.setCellValue("Week Off");
                    cell.setCellStyle(weekendStyle);
                    continue;
                    }

                    String status = allAttendance.stream()
                    .filter(a ->
                    a.getUser().getId()
                    .equals(emp.getId())
                    && a.getDate()
                    .equals(finalDate))
                    .findFirst()
                    .map(a -> a.getStatus().name()
                    .replace("_", " "))
                    .orElse("ABSENT");

                    cell.setCellValue(status);

                    switch (status) {
                    case "PRESENT" ->
                    cell.setCellStyle(presentStyle);
                    case "LATE" ->
                    cell.setCellStyle(lateStyle);
                    case "ABSENT" ->
                    cell.setCellStyle(absentStyle);
                    case "ON LEAVE" ->
                    cell.setCellStyle(leaveStyle);
                    case "WFH" ->
                    cell.setCellStyle(wfhStyle);
                    default ->
                    cell.setCellStyle(absentStyle);
                    }
                    }
                    }

                    if (deptTeam.isEmpty()) {
                    Row emptyRow = sheet.createRow(1);
                    emptyRow.createCell(0).setCellValue(
                    "No employees in this department");
                    }
                    }

                    workbook.write(response.getOutputStream());
                    }
                    }

                    private String escape(String value) {
                    if (value == null) return "";
                    if (value.contains(",") || value.contains("\""))
                    return "\"" + value.replace("\"", "\"\"") + "\"";
                    return value;
                    }
